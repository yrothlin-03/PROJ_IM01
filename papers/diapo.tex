\section{Introduction}

\begin{frame}{Context}
    \begin{itemize}
        \item Motion blur occurs when camera or scene motion affects the exposure.
        \item Variational and Bayesian approaches historically dominate blind deblurring. \cite{MAP_bay}
        \item Deep learning achieves strong results but requires large datasets and heavy computation.
        \item We focus on an interpretable, training-free variational pipeline based on spectral priors.\cite{gf}
    \end{itemize}
\end{frame}

\begin{frame}{Objectives}
    \begin{itemize}
        \item Estimate the blur kernel from the blurred image using anisotropic power spectrum modelling.
        \item Recover the sharp image through TV-regularised deconvolution solved by Split-Bregman.
        \item Pipeline:
        \begin{itemize}
            \item spectral modulus estimation of the kernel,
            \item phase reconstruction under support and positivity constraints,
            \item TV-based restoration with tuned hyperparameters.
        \end{itemize}
    \end{itemize}
\end{frame}

\section{Mathematical Model}

\begin{frame}{Blur Model}
    On considère :
    \[
    v = u * k + b
    \]
    \begin{itemize}
        \item $u$: sharp latent image.
        \item $k$: non-negative, normalized blur kernel with compact support.
        \item $b$: additive impulse noise.
        \item Ill-posed inverse problem requiring prior assumptions.
        \item We assume spatially invariant linear motion blur.
    \end{itemize}
\end{frame}


\section{Blind Kernel Estimation}

\begin{frame}{Prior}
\[
| \widehat{u}(\xi n_\theta) |^2 \approx \frac{1}{\varepsilon + c_\theta \|\xi\|^2}
\]
\[
\Rightarrow | \widehat{D_\theta u}(\xi n_\theta) |^2 = |\xi|^2 \, | \widehat{u}(\xi n_\theta)|^2 = \frac{|\xi|^2}{\varepsilon + c_\theta \|\xi\|^2}
\]
\begin{itemize}
    \item Natural images follow an anisotropic power-law decay.
    \item Taking directional derivatives amplifies high frequencies.
    \item This allows isolating the blur kernel in the spectrum of $D_\theta v$.
\end{itemize}
\end{frame}

\begin{frame}{Whitening and Directional Derivatives}
    \[
    D_\theta v = (D_\theta u) * k + D_\theta b
    \]
    \[
    \widehat{D_\theta v}(\xi n_\theta) = i(\xi n_\theta \cdot n_\theta)\, \widehat{u}(\xi n_\theta)\, \widehat{k}(\xi n_\theta) + i(\xi n_\theta \cdot n_\theta)\,\hat{b}(\xi n_\theta)
    \]
    \begin{itemize}
        \item Whitening: $D_\theta$ flattens the spectrum of natural images.
        \item The remaining directional variations come from the blur kernel.
        \item Enables isolating $|\widehat{k}|^2$ in the autocorrelation domain.
    \end{itemize}

\end{frame}

\begin{frame}{Autocorrelation Analysis}
    Using Parseval's theorem:
    \[
        F\big(R(D_\theta v)\big)(\xi n_\theta) = |\widehat{D_\theta v}(\xi n_\theta)|^2.
    \]
    From the anisotropic power-law prior:
    \[
        |\widehat{D_\theta v}(\xi n_\theta)|^2 = c_\theta |\widehat{k}(\xi n_\theta)|^2 + \mu_\theta + \xi^2|\hat{b}(\xi n_\theta)|^2.
    \]
    \begin{itemize}
        \item The whitened spectrum becomes affine in $|\widehat{k}|^2$.
        \item $\mu_\theta$ is obtained from the kernel normalisation constraint:
        \[
            \int_\mathbb{R} P_\theta(h)(x)\,dx = 1.
        \]
        \item This allows estimating $|\widehat{k}(\xi)|^2$ for all angles $\theta$.
    \end{itemize}
\end{frame}


\begin{frame}{Autocorrelation Deconvolution with compensation filter}
    After estimating the affine model
    \[
        F(R(D_\theta v))(\xi) = c_\theta |\widehat{k}(\xi n_\theta)|^2 + \mu_\theta,
    \]
    the goal is to retrieve a sharper estimate of the directional autocorrelation.

    \textbf{Autocorrelation relation:}
    \[
        R(D_\theta v) = c_\theta * R_k + \mu_\theta + .
    \]

    \begin{itemize}
        \item The width of $R(D_\theta v)$ encodes the blur extent along direction $\theta$.
        \item A compensation filter refines this autocorrelation by removing intrinsic camera blur and noise. 
        \item The correction is computed by deconvolving $R(D_\theta v)$ with a small symmetric kernel via conjugate–gradient descent.
        \item If the corrected autocorrelation becomes non‑positive near the origin, the compensation is rejected to avoid artefacts.
    \end{itemize}
\end{frame}

\begin{frame}{Initial Support Estimation}
    \begin{itemize}
        \item Initial support $s_\theta$ estimated from minima of compensated autocorrelation.
        \item Support is smoothed using the Lipschitz constraint:
        \[
        |s_{\theta_i} - s_{\theta_j}| \le \kappa |\theta_i - \theta_j|.
        \]
        \item Ensures coherent support across all projection directions.
    \end{itemize}
\end{frame}

\begin{frame}{Power spectrum estimation}
    \[
    F(R(D_\theta v))(\xi) = c_\theta |\widehat{k}(\xi n_\theta)|^2 + \mu_\theta
    \]
    \[
    \Rightarrow |\widehat{k}(\xi n_\theta)|^2 = \frac{F(R(D_\theta v))(\xi) - \mu_\theta}{c_\theta}
    \]
    \begin{itemize}
        \item Full modulus $|\widehat{k}(\xi)|$ recovered by interpolating directional slices.
        \item Noise bias removed using low-frequency estimates of $\mu_\theta$.
        \item Produces a stable modulus usable for phase retrieval.
    \end{itemize}
\end{frame}

\begin{frame}{Phase retrieval and kernel estimation}
    \begin{itemize}
        \item Random phases $\phi(\xi)$ sampled to reconstruct candidate kernels:
        \[
        k = \mathcal{F}^{-1}\left(|\widehat{k}(\xi)| e^{i\phi(\xi)}\right)
        \]
        \item Constraints applied: positivity, compact support, normalization.
        \item Best candidate chosen by TV-deconvolution by scoring on the estimated kernel and its symetric version.
    \end{itemize} 
\end{frame}

\begin{frame}{Limitations of Kernel Estimation Method}
    \begin{itemize}
        \item Assumes spatially invariant blur; struggles with non-uniform blur.
        \item Sensitive to noise; high noise can degrade kernel estimates.
        \item Relies on accurate prior model; deviations can lead to errors.
        \item Phase retrieval may yield suboptimal kernels due to random initialization.
        \item Computationally intensive due to multiple iterations and Fourier transforms.
    \end{itemize}
\end{frame}




\section{TV–deconvolution}

\begin{frame}{TV-deconvolution problem}
    We solve the TV-regularised deconvolution problem:
    \[
        \hat{u} = \arg\min_u \; \|k * u - v\|_2^2 + \lambda \|\nabla u\|_1.
    \]
    Using the split-Bregman method (\cite{goldsb}, \cite{tvipol}), introduce an auxiliary variable $d$ enforcing $d = \nabla u$ such that : 
    \[
        (\hat{u}, \hat{d}) = \arg\min_{u,d} \; \|k * u - v\|_2^2 + \lambda \|d\|_1 + \frac{\gamma}{2} \|d - \nabla u\|_2^2.
    \]
    \begin{itemize}
        \item $\lambda$: regularization parameter balancing data fidelity and smoothness.
        \item $\gamma$: penalty parameter enforcing $d \approx \nabla u$.
    \end{itemize}

\end{frame}

\begin{frame}{Split-Bregman method}
    \textbf{Iterative updates:}
    \begin{itemize}
        \item \textbf{$u$-update:} closed-form in Fourier domain (with symmetric boundary conditions)
        \[
            u^{k+1} = \mathcal{F}^{-1}\left( \frac{\frac{\lambda}{\gamma} \overline{\mathcal{F}(k)}\,\mathcal{F}(\text{E}v) - \mathcal{F}(\text{E}\operatorname{div}(d^k - b^k))}{\frac{\lambda}{\gamma}|\mathcal{F}(k)|^2 - \mathcal{F}(\Delta)} \right).
        \]
        \item \textbf{$d$-update:} soft-thresholding (shrinkage)
        \[
            d^{k+1} = \operatorname{shrink}(\nabla u^{k+1} + b^k,\; 1/\gamma).
        \]
        \item \textbf{Bregman update:}
        \[
            b^{k+1} = b^k + (\nabla u^{k+1} - d^{k+1}).
        \]
    \end{itemize}

    \textbf{Stopping criterion:}
    \[
        \|u^{k+1} - u^k\|_2 < \eta.
    \]
    TV preserves edges and reduces ringing, while boundary conditions strongly affect artefacts.
\end{frame}

\begin{frame}{Limitations of TV–deconvolution}
    \begin{itemize}
        \item TV regularization can lead to staircasing and border artefacts effects in smooth regions.
        \item Choice of $\lambda$ and $\gamma$ is critical for balancing noise reduction and detail preservation and needs tuning via cross-validation
        \item Computationally intensive due to iterative nature; convergence speed depends on parameters. (However in practice it converges in ~10 iterations)
        \item Sensitive to kernel estimation errors; inaccurate kernels degrade deconvolution quality.
        \item Sensitive to noise; high noise levels can lead to suboptimal results despite TV regularization.
    \end{itemize}
    
\end{frame}

\begin{frame}{Possible improvements}
    \begin{itemize}
        \item Using hyperlaplacian priors for better edge preservation. (\cite{hyperlaplacian}, \cite{hyperlap_blind})
        \item Pre-denoising (TV-Denoising) before autocorrelation / regularization for high frequencies; (noise reduction).
        \item Estimating support by a more robust method (second derivatives for instances).
        \item Using a more advanced phase retrieval method (e.g., Gerchberg-Saxton algorithm).
        \item Adaptative adjustment of $\alpha$ in compensation filter.
    \end{itemize}
\end{frame}
