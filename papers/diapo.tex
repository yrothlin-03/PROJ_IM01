\section{Introduction}

\begin{frame}{Context and Objectives}

\textbf{Context}
\begin{itemize}
    \item Motion blur arises from camera or scene motion during exposure.
    \item Historically : variational or MAP / bayesian approaches. More recently deep-learning approaches perform well.~\cite{MAP_bay}.
    \item Our work focuses on an interpretable, training-free variational pipeline based on spectral priors~\cite{gf}.
\end{itemize}

\vspace{0.3cm}

\textbf{Objectives}
\begin{itemize}
    \item Estimate the blur kernel directly from the blurred image using anisotropic spectral modelling.
    \item Restore a sharp image via TV-regularised deconvolution solved with Split-Bregman.
    \item Overall pipeline:
    \begin{itemize}
        \item estimation of the kernel's spectral modulus,
        \item phase reconstruction under positivity and support constraints,
        \item TV-based restoration with tuned hyperparameters.
    \end{itemize}
\end{itemize}

\end{frame}

\section{Mathematical Blur Model}

\begin{frame}{Blur Model}

\small
\textbf{Image formation model}
\[
    v \;=\; u * k \;+\; b
\]

\begin{itemize}
    \item $u$: underlying sharp image.
    \item $k$: blur kernel (non-negative, normalized, compact support).
    \item $b$: additive impulse noise.
\end{itemize}

\vspace{0.2cm}

\textbf{Remarks}
\begin{itemize}
    \item Inversion is ill-posed → requires prior assumptions.
    \item We assume a spatially invariant, linear motion blur.
\end{itemize}

\end{frame}



\section{Blind Kernel Estimation}

\begin{frame}{Spectral Prior}

\small
\textbf{Natural image prior (anisotropic power-law decay):}
\[
|\widehat{u}(\xi n_\theta)|^2 
\;\approx\; 
\frac{1}{\varepsilon + c_\theta \|\xi\|^2}
\]

\textbf{Effect of directional derivatives:}
\[
|\widehat{D_\theta u}(\xi n_\theta)|^2 
= |\xi|^2 \,|\widehat{u}(\xi n_\theta)|^2
\;=\;
\frac{|\xi|^2}{\varepsilon + c_\theta \|\xi\|^2}
\]

\vspace{0.2cm}

\textbf{Interpretation}
\begin{itemize}
    \item Natural images exhibit a directional (anisotropic) spectral decay.
    \item Directional derivatives act as a high-frequency amplifier.
    \item This amplification makes the blur kernel more distinguishable  
          in the spectrum of $D_\theta v$, enabling its estimation.
\end{itemize}

\end{frame}

\begin{frame}{Autocorrelation Analysis}
    From the anisotropic power-law prior:
    \[
        |\widehat{D_\theta v}(\xi n_\theta)|^2 = c_\theta |\widehat{k}(\xi n_\theta)|^2 + \mu_\theta + \xi^2|\hat{b}(\xi n_\theta)|^2.
    \]
    Using Parseval's theorem:
    \[
        F\big(R(D_\theta v)\big)(\xi n_\theta) = |\widehat{D_\theta v}(\xi n_\theta)|^2.
    \]
    we obtain : 
    \[
        F(R(D_\theta v))(\xi) = c_\theta |\widehat{k}(\xi n_\theta)|^2 + \mu_\theta,
    \]
    \begin{itemize}
        \item The width of $R(D_\theta v)$ encodes the blur extent along direction $\theta$.
        \item A compensation filter refines this autocorrelation by removing intrinsic camera blur and noise. (by deconvolving with gradient descent)
    \end{itemize}
\end{frame}

\begin{frame}{Initial Support Estimation}
    \begin{itemize}
        \item Initial support $s_\theta$ estimated from minima of compensated autocorrelation.
        \item Support is smoothed using the Lipschitz constraint:
        \[
        |s_{\theta_i} - s_{\theta_j}| \le \kappa |\theta_i - \theta_j|.
        \]
        \item Ensures coherent support across all projection directions.
    \end{itemize}

\begin{frame}{Phase retrieval and kernel estimation}
    \begin{itemize}
        \item Random phases $\phi(\xi)$ sampled to reconstruct candidate kernels:
        \[
        k = \mathcal{F}^{-1}\left(|\widehat{k}(\xi)| e^{i\phi(\xi)}\right)
        \]
        \item Constraints applied: positivity, compact support, normalization.
        \item Best candidate chosen by TV-deconvolution by scoring on the estimated kernel and its symetric version.
    \end{itemize} 
\end{frame}

\begin{frame}{Motivations and Limitations of the Goldstein--Fattal Method}

\textbf{Motivations}
\begin{itemize}
    \item Spectral modelling provides easily interpretable (but limited), training-free .
    \item Works well for spatially invariant, linear motion blur without noise.
\end{itemize}

\vspace{0.2cm}

\textbf{Limitations}
\begin{itemize}
    \item Assumes spatially invariant blur → fails for more complex motion blur.
    \item Sensitive to noise; strong noise deteriorates autocorrelation.
    \item Strongly depends on the validity of the spectral prior; deviations reduce accuracy.
    \item Phase retrieval relies on random initialization → potential suboptimal estimator.
    \item Computationally expensive due to repeated iterations and Fourier-domain operations.
\end{itemize}

\end{frame}

\begin{frame}{TV-deconvolution via Split-Bregman}

\small
\textbf{TV-regularised deconvolution:}
\[
    \hat{u} = \arg\min_u \; \|k * u - v\|_2^2 + \lambda \|\nabla u\|_1.
\]

\textbf{Split Bregman :} Introduce auxiliary variable $d$ and penalty $\gamma$:
\[
    (\hat{u}, \hat{d}) =
    \arg\min_{u,d} \; \frac{\lambda}{2}\|k * u - v\|_2^2 +
     \|d\|_1 + \frac{\gamma}{2}\|d - \nabla u\|_2^2.
\]

\vspace{0.2cm}
\centering

\[
    \|u^{k+1} - u^k\|_2 < \eta.
\]

\begin{frame}{Explanations and Limitations of TV–deconvolution}

\textbf{Why TV-regularisation?}
\begin{itemize}
    \item The $\ell_1$–norm on the gradient ($\|\nabla u\|_1$) preserves sharp edges and avoids over-smoothing.
    \item TV promotes piecewise-smooth regions, leading to clean and stable reconstructions.
    \item The parameter $\lambda$ controls the trade-off between data fidelity and regularisation
          (\(\lambda \uparrow \Rightarrow\) stronger smoothing; \(\lambda \downarrow \Rightarrow\) sharper but noisier).
\end{itemize}

\vspace{0.2cm}
\textbf{Limitations}
\begin{itemize}
    \item TV may introduce \textbf{staircasing} and \textbf{ringing artefacts}, especially around edges.
    \item The parameters $\lambda$ and $\gamma$ must be tuned carefully for stability and convergence.
    \item The method is iterative and can be computationally demanding  
          (although in practice Split-Bregman converges in $\sim$10 iterations).
    \item Performance is sensitive to errors in the estimated blur kernel and to noise in the data.
\end{itemize}

\end{frame}

\begin{frame}{Possible improvements}
    \begin{itemize}
        \item Using hyperlaplacian priors for better edge preservation. (\cite{hyperlaplacian}, \cite{hyperlap_blind})
        \item Pre-denoising (TV-Denoising) before autocorrelation / regularization for high frequencies; (noise reduction).
        \item Estimating support by a more robust method (second derivatives for instances).
        \item Using a more advanced phase retrieval method (e.g., Gerchberg-Saxton algorithm).
        \item Adaptative adjustment of $\alpha$ in compensation filter.
    \end{itemize}
\end{frame}
